CREATE PROCEDURE sp_InsertNewEmployee
@PersonID int ,
@LibraryId int ,
@ManagerId int ,
@IsActive bit,
@HireDate date,
@ContractEndDate date,
@InsedtedID int output
as
begin
 SET NOCOUNT ON;
 insert into Employees(PersonID,LibraryId ,ManagerId ,IsActive,HireDate,ContractEndDate) 
 values (@PersonID,@LibraryId ,@ManagerId ,@IsActive,@HireDate,@ContractEndDate);
 set @InsedtedID =  SCOPE_IDENTITY();
end;
GO

CREATE PROCEDURE sp_UpdateEmployee
@EmployeeID int ,
@IsActive bit,
@HireDate date,
@ContractEndDate date

as
begin
  update Employees 
  set
        IsActive = @IsActive ,
        HireDate = @HireDate ,
        ContractEndDate= @ContractEndDate 
		where EmployeeID = @EmployeeID ;
end

GO

CREATE TRIGGER [dbo].[TR_Employee_InsteadOfDelete]
ON [dbo].Employees INSTEAD OF DELETE
as
 begin

    SET NOCOUNT ON

    update Employees  set IsDeleted = 1 from Employees
    inner join Deleted on Deleted.EmployeeID = Employees.EmployeeID ;
   

    update People  set IsDeleted = 1 from People
    inner join Deleted on Deleted.PersonID = People.PersonID 
   

 end

 GO

 CREATE OR ALTER PROCEDURE SP_DeleteEmployee
    @EmployeeID       INT,
    @IsSuccess      BIT OUTPUT
AS
BEGIN
    BEGIN TRY
        IF NOT EXISTS (SELECT 1 FROM Employees WHERE  EmployeeID  =  @EmployeeID )
        BEGIN
            SET @IsSuccess = 0;
            RETURN;
        END
		SET @IsSuccess = 1; -- because we have instead on delete
            
			
	    DELETE FROM  Employees  WHERE EmployeeID  =  @EmployeeID;
         	       
    END TRY
    BEGIN CATCH
        EXEC SP_SaveErrors;
        THROW;
    END CATCH
END

GO

CREATE OR ALTER PROCEDURE SP_GetEmployeeByID
    @EmployeeID INT
AS
BEGIN
    BEGIN TRY
        SELECT * FROM Employees
        WHERE EmployeeID = @EmployeeID AND IsDeleted = 0;
    END TRY
    BEGIN CATCH
        EXEC SP_SaveErrors;
        THROW;
    END CATCH
END
